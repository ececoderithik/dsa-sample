/* WAP to implement a stack of integers using an array.
   Menu: push, pop, display. Implement isFull(), isEmpty(). */

#include <stdio.h>
#include <stdlib.h>
#define SIZE 5
#define EXIT 0

typedef struct
{
	int stack[SIZE];
	int top;
}STACK;

/****function declarations****/
int isFull(STACK *s);
int isEmpty(STACK *s);
void push(STACK *s, int value);
void pop(STACK *s);
void display(STACK *s);

int main()
{
	STACK s;
	int choice, value;
	s.top = -1;

	do
	{
		printf("\n0.exit\n1.push\n2.pop\n3.display\n");
		printf("give the choice:\t");
		scanf("%d", &choice);

		switch(choice)
		{
			case 1:
				printf("give value:\t");
				scanf("%d", &value);
				push(&s, value);
				break;
			case 2: pop(&s); break;
			case 3: display(&s); break;
			case EXIT: printf("exiting\n"); break;
			default: printf("invalid choice\n");
		}
	}while(choice != EXIT);

	return 0;
}

int isFull(STACK *s)
{
	if(s->top == SIZE - 1)
		return 1;
	else
		return 0;
}

int isEmpty(STACK *s)
{
	if(s->top == -1)
		return 1;
	else
		return 0;
}

void push(STACK *s, int value)
{
	if(isFull(s))
	{
		printf("stack overflow\n");
		return;
	}
	s->top++;				/* move, then store */
	*(s->stack + s->top) = value;
}

void pop(STACK *s)
{
	if(isEmpty(s))
	{
		printf("stack underflow\n");
		return;
	}
	printf("popped: %d\n", *(s->stack + s->top));	/* read, then move */
	s->top--;
}

void display(STACK *s)
{
	if(isEmpty(s))
	{
		printf("stack is empty\n");
		return;
	}
	for(int *q = s->stack + s->top; q >= s->stack; q--)
		printf("%d\t", *q);
	printf("\n");
}
